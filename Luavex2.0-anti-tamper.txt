local function outer_decode(strings, order, OFF, SUB)
  local s = {}
  local p = 0
  local function step_pair(x)
    for i = 1, #x, 2 do
      local u = string.find(ALPH92, string.sub(x, i, i), 1, true)
      local v = string.find(ALPH92, string.sub(x, i + 1, i + 1), 1, true)
      if not u or not v then
        return fail_nil()
      end
      local d = (u - 1) * 92 + v - 1
      p = p + 1
      s[(p + OFF - 1) % 18 + 1] = string.char((d - SUB) % 256)
    end
    return true
  end
  local function step_triplet(x, f)
    for i = 1, #x, 3 do
      local c1, c2, c3 = string.byte(x, i, i + 2)
      if not c3 then
        return fail_nil()
      end
      local d = (c1 - 224) * 4096 + (c2 - 128) * 64 + c3 - 128
      local b
      if f == 0 then
        b = 19968
      elseif f == 1 then
        b = 10240
      else
        if d >= 19968 then
          b = 19968
        elseif d >= 9728 then
          b = 9728
        elseif d >= 9472 then
          b = 9472
        else
          return fail_nil()
        end
      end
      p = p + 1
      s[(p + OFF - 1) % 18 + 1] = string.char((d - b - SUB) % 256)
    end
    return true
  end
  for _, entry in ipairs(order) do
    local x = strings[entry.name]
    local ok
    if entry.f == 3 then
      ok = step_pair(x)
    else
      ok = step_triplet(x, entry.f)
    end
    if not ok then
      return fail_nil()
    end
  end
  local e = table.concat(s, "", 1, 5)
  local m = table.concat(s, "", 6)
  return e, m
end

local function behavioral_canary(E)
  local K = E.bxor
  local L = E.band
  local F = E.floor
  return K(0, 0) == 0
    and L(255, 15) == 15
    and F(1.9) == 1
end

local function table_sum_32640(t)
  local acc = 0
  for i = 1, 256 do
    acc = acc + t[i]
  end
  return acc % 65536 == 32640
end

local function xor_fold(bxor, arr)
  local acc = 0
  for i = 1, #arr do
    acc = bxor(acc, arr[i])
  end
  return acc
end

local function adler32(data)
  local a, b = 1, 0
  for i = 1, #data do
    a = (a + string.byte(data, i)) % 65521
    b = (b + a) % 65521
  end
  return b * 65536 + a
end
