local function xorshift32(a)
        a = band(a, 4294967295)
        a = bxor(a, lshift(a, 13))
        a = bxor(a, rshift(a, 17))
        a = bxor(a, lshift(a, 5))
        return band(a, 4294967295)
end

local function mix2(fM, P)
        local Wi = band(685710331 + fM * 34141 + (fM % 251) * 23047, 4294967295)
        local Lz = band(P * 16843009, 4294967295)
        Wi = bxor(Wi, Lz)
        Wi = bxor(Wi, lshift(Wi, 13))
        Wi = bxor(Wi, rshift(Wi, 17))
        Wi = bxor(Wi, lshift(Wi, 5))
        return band(Wi, 4294967295)
end

local function probeMetatableLock()
        local getmetatable = getfenv()["getmetatable"]
        local reported = getmetatable(string.char())
        if reported ~= "The metatable is locked" then
                return 196
        end
        return 171
end

local function probeNativeStringByte()
        local ok, result = pcall(setfenv, string.byte, getfenv(string.byte))
        if ok or result ~= "'setfenv' cannot change environment of given object" then
                return 196
        end
        return 171
end

local function probeTracebackRepetition()
        local debugLib = getfenv(0)["debug"]
        local marker = string.char(82, 72, 75, 77, 79)
        local function recurse(depth)
                if depth == 0 then
                        return debugLib["traceback"](marker)
                end
                return recurse(depth - 1)
        end
        local trace = recurse(2)
        local counts = {}
        for digits in string.gmatch(trace, ":(%d+)[^\n]*") do
                local line = tonumber(digits)
                if line ~= nil then
                        counts[line] = (counts[line] or 0) + 1
                        if counts[line] > 2 then
                                return 57
                        end
                end
        end
        return 31
end

local function probeBuiltinSanity()
        local xorProbe = bxor(26864, 13230)
        local roundtrip = string.byte(string.char(94), 1, 1)
        local arity = select("#", 23390, 94, 23390)
        if xorProbe == 23390 and roundtrip == 94 and arity == 3 then
                return 63
        end
        return bxor(63, 103)
end

local function decodeBase85(chunk, alphabet)
        local Dg = {}
        for i = 1, #alphabet do
                Dg[string.sub(alphabet, i, i)] = i - 1
        end
        local groups = #chunk // 5
        local out = buffer.create(groups * 4)
        for g = 0, groups - 1 do
                local d = {}
                for j = 0, 4 do
                        local ch = string.sub(chunk, 5 * g + j + 1, 5 * g + j + 1)
                        d[j] = (Dg[ch] - ((5 * g + j) * 41) % 85) % 85
                end
                local n = ((((d[1] * 85 + d[3]) * 85 + d[4]) * 85 + d[2]) * 85 + d[0]) % 4294967296
                buffer.writeu32(out, g * 4, n)
        end
        return out, groups * 4
end

local function assembleCodeBuffer(headerKey, chunkTable, indexStream, seedKey, alphabet)
        local code = buffer.create(9336)
        local offset = 0
        local rolling = seedKey
        for i = 1, #indexStream do
                local index = bxor(
                        bxor(indexStream[i], 40),
                        bxor((172 + i * 173 + (i % 251) * 24) % 256, headerKey)
                )
                local chunk = bxor(index, rolling)
                rolling = band(rolling + chunk * 211 + i * 194, 255)
                indexStream[i] = nil
                if chunk > #chunkTable then
                        chunk = chunk - #chunkTable
                end
                local entry = chunkTable[chunk]
                if entry == nil then
                        return nil
                end
                local decoded, size = decodeBase85(entry, alphabet)
                buffer.copy(code, offset, decoded, 0, size)
                offset = offset + size
        end
        return code
end

local function descrambleSwapXor(code, sessionKey)
        local n = 0
        while n + 1 < 9336 do
                local a = bxor(buffer.readu8(code, n), (sessionKey + 1919273834 + (n + 1) * 31035) % 256)
                local b = bxor(buffer.readu8(code, n + 1), (sessionKey + 1919273834 + (n + 2) * 31035) % 256)
                buffer.writeu8(code, n, b)
                buffer.writeu8(code, n + 1, a)
                n = n + 2
        end
        return xorshift32(sessionKey + 1792098631 + 24827)
end

local function deriveKeystreamTable(seedTable, kh)
        local nQ = band(kh * 17 + 1, 255)
        local eB = band(kh * 31 + 1, 255)
        local LE = {}
        for i = 1, #seedTable do
                LE[i] = band(bxor(bxor(seedTable[i], 40), band(kh + i * nQ + (i % 251) * eB, 255)), 255)
        end
        local hu = bxor(146, band(kh + 253 * nQ + (253 % 251) * eB, 255))
        local aq = bxor(150, band(kh + 254 * nQ + (254 % 251) * eB, 255))
        local Hr = bxor(85, band(kh + 255 * nQ + (255 % 251) * eB, 255))
        return LE, hu, aq, Hr
end

local function descrambleKeyStream(code, LE, hu, aq, Hr, nW)
        local ue = #LE
        for i = 0, 9335 do
                local gr = (i % ue) + 1
                local na = i + 1
                local ud = (hu + na * aq + (na % 251) * Hr) % 256
                local qs = bxor(LE[gr], ud)
                local v = bxor(buffer.readu8(code, i), (nW + 1463987739 + na * 11506) % 256)
                buffer.writeu8(code, i, bxor(v, qs))
        end
        return xorshift32(nW + 1850069598 + 112710)
end

local function descrambleRotate(code, sessionKey)
        for i = 0, 9335 do
                local uo = bxor(buffer.readu8(code, i), (sessionKey + 1713177291 + (i + 1) * 27938) % 256)
                buffer.writeu8(code, i, band(bor(rshift(uo, 3), lshift(uo, 5)), 255))
        end
        return xorshift32(sessionKey + 2131166046 + 116694)
end

local function extractPayload(buf)
        local pos = 4
        while pos < 8364 do
                local tag = buffer.readu8(buf, pos)
                pos = pos + 1
                if tag == 3 then
                        pos = pos + 1
                elseif tag == 238 then
                        pos = pos + 1 + buffer.readu8(buf, pos)
                elseif tag == 154 then
                        local len = buffer.readu32(buf, pos)
                        pos = pos + 4
                        if len ~= 8188 then
                                return nil
                        end
                        local out = buffer.create(len)
                        buffer.copy(out, 0, buf, pos, len)
                        return out
                else
                        return nil
                end
        end
        return nil
end

local function makePermutation(kh)
        local ZO = bxor(3185490210, mix2(58, kh)) % 9336
        local cl = bxor(1286092070, mix2(382, kh)) % 9336
        return function(F)
                return ((F - 1) * ZO + cl) % 9336
        end
end

local function makeStreamReader(frame)
        return function()
                local OM = frame.fX
                local cE = buffer.readu8(frame.Fr, OM - 1)
                local hh = frame.TY
                local of = band(hh + 1406318100 + OM * 56314, 4294967295)
                of = bxor(of, 1759978801)
                of = bxor(of, lshift(of, 13))
                of = bxor(of, rshift(of, 17))
                of = bxor(of, lshift(of, 5))
                of = band(of, 4294967295)
                local tm = band(bxor(bxor(bxor(of, rshift(of, 8)), rshift(of, 16)), rshift(of, 24)), 255)
                local NE = band(bor(rshift(cE, 6), lshift(cE, 2)), 255) - 214
                if NE < 0 then
                        NE = NE + 256
                end
                local TS = bxor(NE, tm)
                local gM = band(hh * 40849 + 975607815 + OM * 32609 + TS * 25579 + cE * 45786, 4294967295)

                gM = bxor(gM, rshift(gM, 17))

                frame.TY = gM
                frame.fX = OM + 1
                return TS

local function decodeProgram(kh, encryptedBlob, alphabet, seedTable)
        local body = string.sub(encryptedBlob, 13)
        local zz = bxor(3476488477, xorshift32(bxor(band(kh * 16843009, 4294967295), 3861205832)))
        local code, size = decodeBase85(body, alphabet)
        if code == nil or size ~= 9336 then
                return nil
        end
        local gn = descrambleSwapXor(code, zz)
        local LE, hu, aq, Hr = deriveKeystreamTable(seedTable, kh)
        local ds = descrambleKeyStream(code, LE, hu, aq, Hr, gn)
        descrambleRotate(code, ds)
        local offsetOf = makePermutation(kh)
        local secondary = buffer.create(8364)
        for uX = 1, 8364 do
                buffer.writeu8(secondary, uX - 1, buffer.readu8(code, offsetOf(972 + uX)))
        end
        local payload = extractPayload(secondary)
        return payload, zz
end

local function reencodeFunctionOpcodes(fn)

        end
        fn[7] = true
        local instructions = fn[1]

        fn[8] = seed
        for index = 1, #instructions do
                local ins = instructions[index]
                local opcode = ins[1]
                local parity = (index * 46 + opcode * 31 + (#instructions + 1) * 24253) % 2
                local ps = (index * 96 + opcode * 36 + (#instructions + 1) * 65036) % 256
                local rt = ((opcode * 31 + index * 46 + #instructions + seed) % 255) + 1
                local pg = (ps + rt) % 256
                ins[5] = ps

                        ins[6] = bxor(opcode, pg)

                        ins[1] = 71040

                end

local function remapThroughPool(raw, pools)
        local selector = raw % 4
        local entry = pools[selector]
        if not entry then
                return nil
        end
        local byte = string.byte(entry.string, (raw - selector) / 4 + 1)
        if not byte then
                return nil
        end
        local value = byte - entry.bias
        if value < 0 then
                value = value + 256
        end
        return value
end

local function checksumModule(module, keyA, keyB)
        local instructions = module[1]
        if type(instructions) ~= "table" then
                return nil
        end
        local acc = 1403030849 + keyA * 26291 + keyB * 25363 + (#instructions + 1) * 9123
        for i = 1, #instructions do
                local rec = instructions[i][32]
                if not rec then
                        return nil
                end
                acc = bxor(acc, rec[3] or 0) + (rec[11] or 0) * 21449 + i * 56509 + (rec[2] or 1744) * 64671
        end
        return band(bxor(acc, (module[5] or 0) * 52127) + (module[3] or 0) * 14173, 4294967295)
end

local function commitModule(module, keyA, keyB, handle, token)
        local checksum = checksumModule(module, keyA, keyB)
        if checksum == nil then
                return nil
        end
        module[475] = bxor(keyA, 540841016)
        module[408] = bxor(keyB, 1115371541)
        module[354] = bxor(checksum, band(1747677602 + keyA * 62773 + keyB * 11659, 4294967295))
        if pending[handle] ~= token then
                return nil
        end
        pending[handle] = nil
        registry[module] = { bxor(module[354], 2269905765), module[5] or 0 }
        return module
end

local function verifyModule(module)
        if type(module) ~= "table" then
                return false
        end
        local keyA = bxor(module[475] or 0, 540841016)
        local keyB = bxor(module[408] or 0, 1115371541)
        local stored = module[354]
        local instructions = module[1]
        if type(instructions) ~= "table" or type(stored) ~= "number" then
                return false
        end
        if keyA < 0 or keyB < 0 or stored == 0 then
                return false
        end
        local entry = registry[module]
        if entry == nil or entry[1] ~= bxor(stored, 2269905765) or entry[2] ~= (module[5] or 0) then
                return false
        end
        return verifyInstruction(instructions[1]) and verifyInstruction(instructions[#instructions])
end

local function decodeProto(store, index, deserialize)
        local raw = store[index]

                local bc = band(context, 255)
                local rk = (bc * 243 + ydValue + 237) % 256
                local token = string.char(bxor(b1, bc)) .. string.char(bxor(b2, rk)) .. string.char(bc) .. string.char(ydValue) .. string.char(rk)
                challengeLedger[token] = true
                return token
        end
        return constantBytes
end

local function installChallengeGlobal(secret)
        getfenv(0)["HX"] = function(eR, wT, mq, HS, Ow, Zf, fF, Si)
                local c3 = string.byte(Si, 3, 3)
                local c4 = string.byte(Si, 4, 4)
                local c5 = string.byte(Si, 5, 5)
                if not c3 or not c4 or not c5 or c5 ~= (c3 * 243 + c4 + 237) % 256 then
                        return bxor(secret, 253)
                end
                local counter = challengeLedger[1226908804]
                if counter == nil or counter <= 0 or challengeLedger[Si] == nil then
                        return bxor(secret, 253)
                end
                local th = (string.byte(eR, 1, 1) * 73 + string.byte(eR, 2, 2) * 151 + wT * 29 + mq * 17 + (HS % 256) * 11 + Ow * 7 + Zf * 37 + fF * 53 + 165) % 256
                return bxor(secret, th)
        end
end

local callDepth = 0

local function protectedCall(fn, ...)
        callDepth = callDepth + 1
        local results = table.pack(pcall(fn, ...))
        callDepth = callDepth - 1
        if not results[1] then
                error(results[2], 0)
        end
        return table.unpack(results, 2, results.n)
end

local function runFrame(frame, nodeGraph, firstNode, opcodePools)
        local d = firstNode
        local registers = frame.registers
        while true do
                local n = nodeGraph[d]
                if n[32] then
                        if not verifyLabelDescriptor(n) then
                                return nil
                        end
                end
                local opcode = n[104]
                if not opcode then
                        opcode = decodeOpcode(n, opcodePools)
                        n[104] = opcode
                end
                if opcode == 510 then
                        if registers[n[2]] ~= n[4] then
                                d = nodeGraph[d][185]
                        else
                                d = n[3]
                        end
                        continue
                elseif opcode == 533 then
                        local base = n[2]
                        local results = table.pack(registers[base](registers[base + 1]))
                        local top = results.n + base - 1
                        local j = 0
                        for r = base, top do
                                j = j + 1
                                registers[r] = results[j]
                        end
                        d = nodeGraph[d][185]
                        continue
                elseif opcode == 526 then
                        return
                end
                d = nodeGraph[d][185]
        end
end

local function bootstrap(vmTable)
        vmTable[545] = getfenv()[string.char(115, 101, 116, 102, 101, 110, 118)]
        if vmTable[913] == nil then
                vmTable[913] = {}
        end
        for byte = 0, 255 do
                vmTable[913][byte] = string.char(byte)
        end
        local entry = vmTable[1284](vmTable)
        local wrapper
        wrapper = function(...)
                local results = entry(...)
                vmTable[195] = nil
                return results
        end
        return wrapper
end

local function mountProtections(vmTable, holder, chunkTable, encryptedAlphabet, seedTable)
        local jD, uE, sG = runEnvironmentProbes()
        local kh, ut = deriveSessionKey(jD, uE, sG)
        if ut ~= 0 then
                onTamperDetected(vmTable, makeReferenceHolder(), chunkTable, ut)
                return nil
        end
        local blob = getBlob(holder)
        if blob == nil then
                return nil
        end
        local payload = decodeProgram(kh, blob, encryptedAlphabet, seedTable)
        if payload == nil then
                return nil
        end
        local frame = makeFrame(payload, kh)
        local reader = makeStreamReader(frame)
        wipeRuntimeState(holder)
        return {
                kh = kh,
                frame = frame,
                reader = reader,
                permutation = makePermutation(kh),
                challengeLedger = challengeLedger,
        }
end
