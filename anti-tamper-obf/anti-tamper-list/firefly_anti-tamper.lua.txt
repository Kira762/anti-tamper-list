local function bxor(a, b)
    local result = 0
    local bit = 1
    while a > 0 or b > 0 do
        local x = a % 2
        local y = b % 2
        if x ~= y then
            result = result + bit
        end
        a = (a - x) / 2
        b = (b - y) / 2
        bit = bit * 2
    end
    return result
end

local function read_le(buffer, position, width)
    local value = 0
    for offset = 0, width - 1 do
        value = value + buffer[position + offset] * 256 ^ offset
    end
    return value
end

function AntiTamper.verify_chunks(chunks, build)
    build = pick(build)

    local total = 0
    for index = 1, #chunks do
        local chunk = chunks[index]
        local text = chunk[3]
        local sum = ((chunk[1] + chunk[2] * 17) + #text * 31) % 65536
        for position = 1, #text do
            sum = (sum + text:byte(position) * (position % 13 + 1)) % 65536
        end
        if sum ~= chunk[4] then
            return nil
        end
        total = (total + sum) % 65536
        if chunk[1] == build.chunk_tag then
            selected[chunk[2]] = text
        end
    end
    if total ~= build.chunk_total then
        return nil
    end
    return selected
end

function AntiTamper.crc16(bytes, build)
    build = pick(build)
    local crc = bxor(65535, build.crc_seed)
    for index = 1, #bytes do
        crc = bxor(crc, bytes[index])
        for _ = 0, 7 do
            if crc % 2 == 1 then
                crc = bxor(math.floor(crc / 2), build.crc_polynomial)
            else
                crc = math.floor(crc / 2)
            end
        end
    end
    return crc
end

function AntiTamper.run(options)
    options = options or {}
    local env = options.env or _ENV or _G
    local build = pick(options.build)
    if options.proxy then
        AntiTamper.install_env_proxy(env)
    end
    local failures = {}
    if AntiTamper.environment_check(env, build) then
        failures[#failures + 1] = "environment"
    end
    if not AntiTamper.runtime_selftest(options) then
        failures[#failures + 1] = "selftest"
    end
    if options.watermark ~= nil and AntiTamper.watermark_check(options.watermark, options.encrypted_watermark, options.watermark_key, build) then
        failures[#failures + 1] = "watermark"
    end
    return #failures == 0, failures
end
