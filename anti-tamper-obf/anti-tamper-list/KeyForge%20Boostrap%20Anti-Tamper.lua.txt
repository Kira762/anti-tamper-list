local function waitForLocalPlayer()
	local Players=game:GetService("Players")
	local plr=Players.LocalPlayer
	if plr then return plr end
	local t0=os.clock()
	while not plr and (os.clock()-t0)<10 do
		pcall(function() RunService.Heartbeat:Wait() end)
		plr=Players.LocalPlayer
	end
	if not plr then
		local ok,res=pcall(function() return Players.PlayerAdded:Wait() end)
		if ok then plr=res end
	end
	return plr
end

local function ensureUiScale(card)

local function fadeChildrenIn(card,delaySec,dur,style)
	local i=0
	for _,ch in ipairs(card:GetDescendants())do
		if ch:IsA("TextLabel")or ch:IsA("TextButton")or ch:IsA("TextBox")or ch:IsA("ImageLabel")then
			i=i+1
			local stagger=(style=="elegant")and(0.04*i)or 0
			if ch:IsA("ImageLabel")then
				local it=ch.ImageTransparency
				ch.ImageTransparency=1
				_kf_t.delay((delaySec or 0.08)+stagger,function()smoothTween(ch,{ImageTransparency=it},dur or 0.4,Enum.EasingStyle.Quint):Play()end)
			else
				local tt=ch.TextTransparency
				ch.TextTransparency=1
				_kf_t.delay((delaySec or 0.08)+stagger,function()smoothTween(ch,{TextTransparency=tt},dur or 0.4,Enum.EasingStyle.Quint):Play()end)
			end
		end
	end
end
local function animateCardIn(card,strokeObj,style)
	style=tostring(style or "modern")
	card.BackgroundTransparency=1
	if strokeObj then strokeObj.Transparency=1 end

	local scale=ensureUiScale(card)
	if style=="fade" then

local function copyToClipboard(text)
	if type(text)~="string" or text=="" then return false end
	local copied=false
	pcall(function() if setclipboard then setclipboard(text) copied=true end end)
	pcall(function() if not copied and toclipboard then toclipboard(text) copied=true end end)
	pcall(function() if not copied and syn and type(syn.write_clipboard)=="function" then syn.write_clipboard(text) copied=true end end)
	pcall(function() if not copied and syn and type(syn.setclipboard)=="function" then syn.setclipboard(text) copied=true end end)
	return copied
end
local function keySavePath(storageId)
	local id=tostring(storageId or "default"):gsub("[^%w%-_]",""):sub(1,80)
	return "keyforge/keys/"..id..".txt"
end
local function loadSavedKey(storageId)
	local path=keySavePath(storageId)
	local ok,val=pcall(function()
		if not (isfile and readfile and isfile(path)) then return nil end
		local raw=readfile(path)
		if type(raw)~="string" then return nil end
		raw=raw:gsub("^%s+",""):gsub("%s+$","")
		if #raw<4 then return nil end
		return raw
	end)
	return ok and val or nil
end
local function saveKey(storageId,key)
	if type(key)~="string" or #key<4 then return end
	local path=keySavePath(storageId)
	pcall(function()
		if not writefile then return end
		if makefolder then pcall(makefolder,"keyforge") pcall(makefolder,"keyforge/keys") end
		writefile(path,key)
	end)
end
local function clearSavedKey(storageId)
	local path=keySavePath(storageId)
	pcall(function()
		if delfile and isfile and isfile(path) then delfile(path) end
	end)
end

local function spawnKeyParticles(card,content,style,accent)
	if style=="none" or style=="" or not card or not content then return end
	local layer=Instance.new("Frame")
	layer.Name="KF_Particles"
	layer.BackgroundTransparency=1
	layer.BorderSizePixel=0
	layer.Position=UDim2.fromOffset(0,0)
	layer.Size=UDim2.fromOffset(0,0)
	layer.ClipsDescendants=true
	layer.ZIndex=0
	layer.Parent=card
	local function syncSize()
		local sz=content.AbsoluteSize
		layer.Size=UDim2.fromOffset(math.max(1,math.floor(sz.X+0.5)),math.max(1,math.floor(sz.Y+0.5)))
	end
	syncSize()
	content:GetPropertyChangedSignal("AbsoluteSize"):Connect(syncSize)
	local count=isMobile()and 10 or 16
	local parts={}
	for i=1,count do
		local d=Instance.new("Frame")
		local sz=(style=="stars")and math.random(2,3)or math.random(2,5)
		d.Size=UDim2.fromOffset(sz,sz)
		d.BorderSizePixel=0
		d.BackgroundColor3=accent
		d.BackgroundTransparency=math.random(20,55)/100
		d.ZIndex=0
		d.Parent=layer
		corner(d,99)
		parts[i]={
			f=d,
			x=math.random(),
			y=math.random(),
			sp=(style=="snow"and 0.04 or 0.07)+math.random()*0.08,
			drift=(math.random()-0.5)*0.04,
			phase=math.random()*6.28,
		}
		d.Position=UDim2.fromScale(parts[i].x,parts[i].y)
	end
	local conn
	conn=RunService.RenderStepped:Connect(function(dt)
		if not layer.Parent then if conn then conn:Disconnect() end return end
		for _,p in ipairs(parts)do
			if style=="embers" or style=="sparks" then
				p.y=p.y-p.sp*dt
				p.x=p.x+p.drift*dt
				if p.y<-0.05 then p.y=1.05 p.x=math.random() end
			elseif style=="snow" then
				p.y=p.y+p.sp*dt
				p.x=p.x+math.sin(os.clock()*1.4+p.phase)*0.01*dt*60
				if p.y>1.05 then p.y=-0.05 p.x=math.random() end
			else
				local pulse=0.35+0.45*math.abs(math.sin(os.clock()*2.2+p.phase))
				p.f.BackgroundTransparency=pulse
			end
			if p.x<0 then p.x=1 elseif p.x>1 then p.x=0 end
			p.f.Position=UDim2.fromScale(p.x,p.y)
		end
	end)
end

				elseif lower:find("locked to another")or lower:find("hwid")or lower:find("device identity")or lower:find("device lock")then
					if not lower:find("reset hwid")then msg=msg.." Ã¢â‚¬â€ reset HWID in Discord or use the original device" end
				end
				status.Text=msg
			end
		end)
	end
	verify.MouseButton1Click:Connect(submit)

	animateCardIn(card,cardStroke,"pop")

		status.Text="Saved key found Ã¢â‚¬â€ tap Clear saved key to use a different one"
		_kf_t.delay(2.8,function()
			if cancelAutoSubmit or busy then return end
			submitFromSaved=true
			submit()
		end)
	end

local function kfStripBom(s)
  if type(s)~="string" or s=="" then return s end
  if #s>=3 and s:byte(1)==239 and s:byte(2)==187 and s:byte(3)==191 then s=s:sub(4) end
  if #s>=1 and s:sub(1,1)=="\u{FEFF}" then s=s:sub(2) end
  return (s:gsub("\u{200B}",""):gsub("\u{200C}",""):gsub("\u{200D}",""):gsub("\u{FEFF}",""))
end
local function kfFetch(u)
  local function fromHttpGet()
    local ok,body=pcall(function()
      if type(httpget)=="function" then return httpget(u) end
      if game.HttpGetAsync then return game:HttpGetAsync(u) end
      return game:HttpGet(u,true)
    end)
    if ok and type(body)=="string" and body~="" then return kfStripBom(body) end
    if not ok and type(body)=="string" and body~="" then return nil,body end
    return nil
  end
  local function fromRequest(fn)
    if type(fn)~="function" then return nil end
    local ok,res=pcall(function() return fn({Url=u,Method="GET",Headers={["X-KeyForge-Executor"]="1"}}) end)
    if not ok or type(res)~="table" then return nil end
    local status=res.StatusCode or res.status_code or 0
    local body=res.Body or res.body
    if type(body)~="string" then body=body and tostring(body) or "" end
    body=kfStripBom(body)
    if body~="" and (status==200 or body:find("KeyForge",1,true) or body:find("Keyforge",1,true)) then return body end
    return nil
  end
  if type(Xeno)=="table" then
    local body,err=fromHttpGet()
    if body then return body end
    body=fromRequest(Xeno.request or request)
    if body then return body end
    if err then return nil,err end
  end
  do
    local body=fromRequest((type(Xeno)=="table" and (Xeno.request or request)) or (syn and syn.request) or (http and http.request) or (fluxus and fluxus.request) or request or http_request)
    if body then return body end
  end
  do
    local Hs=game:GetService("HttpService")
    if Hs.RequestAsync then
      local ok,res=pcall(function() return Hs:RequestAsync({Url=u,Method="GET",Headers={["X-KeyForge-Executor"]="1"}}) end)
      if ok and type(res)=="table" then
        local status=res.StatusCode or 0
        local body=res.Body or ""
        if type(body)=="string" then body=kfStripBom(body) end
        if type(body)=="string" and body~="" and status==200 then return body end
      end
    end
  end
  local body,err=fromHttpGet()
  if body then return body end
  return nil,err or "failed to fetch loader"
end
local function kfTryLs(fn)
  if type(fn)~="function" then return nil end
  local ok,res=pcall(fn,"return 1")
  if ok and type(res)=="function" then return fn end
  ok,res=pcall(fn,"return 1","=kf","t")
  if ok and type(res)=="function" then return fn end
  local m=tostring(res or "")
  if m:find("RobloxScript",1,true) or m:find("not available",1,true) then return nil end
  return nil
end
local function kfPickLs(t)
  if type(t)~="table" then return nil end
  return kfTryLs(t.__cload) or kfTryLs(t.loadstring) or kfTryLs(t.load)
end
local function kfResolveGuiLoadstring()
  if type(Xeno)=="table" then
    local ls=kfPickLs(Xeno)
    if ls then return ls end
  end
  if type(getgenv)=="function" then
    local ok,ge=pcall(getgenv)
    if ok then
      local ls=kfPickLs(ge)
      if ls then return ls end
    end
  end

    local ok,re=pcall(getrenv)
    if ok then
      local ls=kfPickLs(re)
      if ls then return ls end
    end
  end
  return kfTryLs(loadstring) or kfTryLs(load)
end

local function fetchBootstrap(key)
  local function kfEncodeKey(s)
    return (tostring(s):gsub("([^%w%._~%-])", function(c)
      return string.format("%%%02X", string.byte(c))
    end))
  end
  local url=KEY_LOAD_URL.."&key="..kfEncodeKey(key).."&_cb="..tostring(os.clock()).."-"..tostring(math.random(0,2147483647))
  local body, kfErr = kfFetch(url)
  if not body then error(tostring(kfErr or "failed to fetch loader"), 0) end
  body = kfStripBom(body)
  local ls=_kf_ls or kfResolveGuiLoadstring()
  assert(type(ls)=="function", "no loader function Ã¢â‚¬â€ update Xeno or try another executor")
  local fn, err = ls(body)
  if type(fn) ~= "function" then error(tostring(err or "invalid loader"), 0) end
  local ge=(type(getgenv)=="function" and (function() local ok,g=pcall(getgenv) return ok and g or nil end)()) or _G
  if type(ge)=="table" then
    ge.__KF_GATE_OK=false
    ge.__KF_GATE_FAIL=false
    ge.__KF_GATE_ERR=nil
  end
  local spawnFn=(task and task.spawn) or spawn or function(f) if type(coroutine)=="table" and type(coroutine.wrap)=="function" then coroutine.wrap(f)() else f() end end
  local waitFn=(task and task.wait) or wait
  spawnFn(function()
    local ok,runErr=pcall(fn)
    if not ok then
      local msg=tostring(runErr or "script failed")
      pcall(function() if type(warn)=="function" then warn("[Keyforge Auth] "..msg) end end)
      pcall(function() if type(print)=="function" then print("[Keyforge Auth] "..msg) end end)
      if type(ge)=="table" and not ge.__KF_GATE_OK and not ge.__KF_GATE_FAIL then
        ge.__KF_GATE_FAIL=true
        ge.__KF_GATE_ERR=msg
      end
    end
  end)
  local deadline=(type(os)=="table" and type(os.clock)=="function" and (os.clock()+45)) or nil
  while true do
    if type(ge)=="table" then
      if ge.__KF_GATE_FAIL then
        error(tostring(ge.__KF_GATE_ERR or "authentication failed"), 0)
      end
      if ge.__KF_GATE_OK then return end
    end
    if deadline and type(os)=="table" and type(os.clock)=="function" and os.clock()>deadline then
      error("authentication timed out Ã¢â‚¬â€ retry in a moment", 0)
    end
    if type(waitFn)=="function" then waitFn(0.05) else break end
  end
end
