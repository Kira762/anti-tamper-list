local trackedReferences = {}

local function protectedLookup(target, key)
    if typeOf(target) ~= "table" then
        return true, nil
    end

    return protectedCall(function()
        return target[key]
    end)
end

local function trackReference(target, key, checkForHook, required)
    if typeOf(target) ~= "table" then
        return
    end

    local succeeded, value = protectedLookup(target, key)
    if succeeded then
        trackedReferences[#trackedReferences + 1] = {
            target = target,
            key = key,
            expectedValue = value,
            checkForHook = checkForHook,
            required = required
        }
    end
end

local function trackTableMember(target, key, required)
    if typeOf(target) ~= "table" then
        return
    end

    local succeeded, value = protectedLookup(target, key)
    if succeeded then
        trackedReferences[#trackedReferences + 1] = {
            target = target,
            key = key,
            expectedValue = value,
            checkForHook = true,
            required = required
        }
    end
end

local function checkEnvironmentIntegrity(runBehaviorChecks)
    if environment.getfenv ~= originalGetfenv then
        return false
    end

    for index = 1, #trackedReferences do
        local reference = trackedReferences[index]
        local expectedValue = reference.expectedValue

        if reference.required and
            (expectedValue == nil or
                (reference.checkForHook and typeOf(expectedValue) ~= "function")) then
            return false
        end

        local succeeded, currentValue =
            protectedLookup(reference.target, reference.key)
        if not succeeded or currentValue ~= expectedValue then
            return false
        end

        if reference.checkForHook and
            typeOf(expectedValue) == "function" and
            typeOf(hookDetector) == "function" then
            local hookCheckSucceeded, isHooked =
                protectedCall(hookDetector, expectedValue)
            if hookCheckSucceeded and isHooked == true then
                return false
            end
        end
    end

    if not runBehaviorChecks then
        return true
    end

    if selectedDebugFunction then
        if usesDebugInfo then
            local succeeded, lineNumber, parameterCount, isVararg, reportedFunction =
                protectedCall(selectedDebugFunction, tupleProbe, "laf")

            if not succeeded or
                typeOf(lineNumber) ~= "number" or
                lineNumber < 0 or
                parameterCount ~= 2 or
                isVararg ~= false or
                reportedFunction ~= tupleProbe then
                return false
            end
        else
            local succeeded, info =
                protectedCall(selectedDebugFunction, tupleProbe, "S")

            if not succeeded or
                typeOf(info) ~= "table" or
                info.what ~= "Lua" or
                typeOf(info.linedefined) ~= "number" or
                info.linedefined < 0 then
                return false
            end
        end
    end

    local librariesPassed, behaviorChecksPassed = protectedCall(function()
        return stringLibrary.byte(stringLibrary.char(0x41, 0x00, 0x5A), 0x02) == 0x00
            and stringLibrary.sub("abc", 0x02, 0x03) == "bc"
            and stringLibrary.reverse("abc") == "cba"
            and tableLibrary.concat({"a", "b"}, ":") == "a:b"
            and mathLibrary.floor(0xF / 0x4) == 0x03
            and environment.tonumber("17") == 0x11
            and environment.tostring(false) == "false"
    end)

    if not librariesPassed or behaviorChecksPassed ~= true then
        return false
    end

    if originalGetfenv(0x01) ~= environment then
        return false
    end

    if typeOf(0x01) ~= "number" or
        typeOf(false) ~= "boolean" or
        typeOf(nil) ~= "nil" or
        typeOf({}) ~= "table" then
        return false
    end

    local callSucceeded, firstValue, middleValue, lastValue =
        protectedCall(tupleProbe, 0x0D, 0x11)
    local returnedValueCount =
        environment.select("#", protectedCall(tupleProbe, 0x0D, 0x11))

    if callSucceeded ~= true or
        firstValue ~= 0x0D or
        middleValue ~= nil or
        lastValue ~= 0x11 or
        returnedValueCount ~= 0x04 then
        return false
    end

    local errorCallSucceeded, actualError = protectedCall(function()
        raiseError(expectedError, 0)
    end)

    if errorCallSucceeded ~= false or actualError ~= expectedError then
        return false
    end

    return true
end
